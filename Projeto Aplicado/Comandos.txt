# No Athena, para criar o banco de dados, basta rodar

CREATE DATABASE IF NOT EXISTS financial_db;

# Para criar a tabela

CREATE EXTERNAL TABLE financial_data (
  data_hora BIGINT,
  ibovespa DOUBLE,
  sp500 DOUBLE,
  dowjones DOUBLE,
  usd_brl DOUBLE,
  eur_brl DOUBLE,
  gbp_brl DOUBLE,
  jpy_brl DOUBLE,
  bitcoin DOUBLE,
  ethereum DOUBLE,
  vix DOUBLE,
  vix_br DOUBLE
)
ROW FORMAT SERDE 'org.openx.data.jsonserde.JsonSerDe'
LOCATION 's3://pa-financial/raw-data/kafka/financial/postgres-financial/partition=0/';

# Convertendo timestamp corretamente

SELECT
  from_unixtime(data_hora/1000) AS time,
  ibovespa,
  sp500,
  bitcoin
FROM financial_data
ORDER BY time DESC
LIMIT 50;

# SQL para criação do dashboard das bolsas de valores

SELECT
  from_unixtime(data_hora / 1000) as time,
  'IBOVESPA' as asset,
  ibovespa as value
FROM financial.financial_data
WHERE $__timeFilter(from_unixtime(data_hora / 1000))

UNION ALL

SELECT
  from_unixtime(data_hora / 1000) as time,
  'SP500' as asset,
  sp500 as value
FROM financial.financial_data
WHERE $__timeFilter(from_unixtime(data_hora / 1000))

UNION ALL

SELECT

  'DOWJONES' as asset,
  dowjones as value

WHERE $__timeFilter(from_unixtime(data_hora / 1000))

ORDER BY time
